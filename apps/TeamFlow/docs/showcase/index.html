// BeforeCreate GORM Hook：创建前自动加密密码，昵称兜底 func (u *User) BeforeCreate(tx *gorm.DB) error { // 1. 密码 bcrypt 加密（cost = 10） hashed, err := bcrypt.GenerateFromPassword([]byte(u.Password), bcrypt.DefaultCost) if err != nil { return err } u.Password = string(hashed) // 2. 昵称为空时使用用户名 if u.Nickname == "" { u.Nickname = u.Username } return nil }
// CreateUser 创建用户，翻译 MySQL 1062 唯一冲突为业务错误 func (s *UserService) CreateUser(username, email, password string) (*model.User, error) { user := &model.User{ Username: username, Email: email, Password: password, } if err := storage.DB.Create(user).Error; err != nil { if isMySQLDuplicate(err) { return nil, AppError{409, "用户名或邮箱已存在"} } return nil, err } return user, nil } // UpdateUser 空密码跳过，避免覆盖已加密密码 func (s *UserService) UpdateUser(id uint, username, email, password string) error { var user model.User if err := storage.DB.First(&user, id).Error; err != nil { return err } user.Username = username user.Email = email if password != "" { user.Password = password // BeforeUpdate 会自动哈希 } return storage.DB.Save(&user).Error }
// ListUsersByTeamID 通过 team_members 关联表查询团队用户 func (r *UserRepository) ListUsersByTeamID(teamID uint, page, pageSize int) ([]model.User, int64, error) { var users []model.User var total int64 // 关联子查询：用户必须在指定团队中 subQuery := storage.DB. Model(&model.TeamMember{}). Select("user_id"). Where("team_id = ?", teamID) err := storage.DB. Model(&model.User{}). Where("id IN (?)", subQuery). Count(&total). Offset((page - 1) * pageSize). Limit(pageSize). Find(&users).Error return users, total, err }
